"""One-time AST-bounded extraction preserving original implementation bodies."""
import ast
from pathlib import Path

root = Path('src/looplane')
source = (root / 'runtime.py').read_text()
lines = source.splitlines(keepends=True)
nodes = ast.parse(source).body
blocks = {}
for node in nodes:
    if isinstance(node, (ast.FunctionDef, ast.ClassDef)):
        start = min([node.lineno, *(d.lineno for d in node.decorator_list)])
        blocks[node.name] = ''.join(lines[start - 1:node.end_lineno])
    elif isinstance(node, ast.Assign):
        for target in node.targets:
            if isinstance(target, ast.Name):
                blocks[target.id] = ''.join(lines[node.lineno - 1:node.end_lineno])

def write(path, imports, names, replacements=()):
    body = '\n\n\n'.join(blocks[name].rstrip() for name in names)
    for old, new in replacements:
        assert old in body, (path, old)
        body = body.replace(old, new)
    target = root / path
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text('from __future__ import annotations\n\n' + imports + '\n\n\n' + body + '\n')

for package in ('execution', 'sandbox', 'workspace'):
    directory = root / package
    directory.mkdir(exist_ok=True)
    init = directory / '__init__.py'
    if not init.exists():
        init.write_text(f'"""Local {package} capabilities."""\n')
